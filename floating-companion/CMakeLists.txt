cmake_minimum_required(VERSION 3.20)
project(FloatingCompanion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(gz-sim REQUIRED)
find_package(gz-plugin REQUIRED COMPONENTS register)
find_package(gz-transport REQUIRED)
if(NOT TARGET jsoncpp_lib)
  find_package(jsoncpp CONFIG REQUIRED)
endif()
add_library(Companion SHARED src/Companion.cc)
target_link_libraries(Companion PRIVATE gz-sim::gz-sim gz-plugin::register gz-transport::gz-transport jsoncpp_lib)
target_compile_options(Companion PRIVATE -Wall -Wextra -Wpedantic)
